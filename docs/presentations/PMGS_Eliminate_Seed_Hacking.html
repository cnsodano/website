read_data_transparently_published_on_osf <- function(identifier){
    project = osfr::osf_retrieve_node(paste0("https://osf.io/", identifier
    ,"/")) |> osfr::osf_ls_files()
    data_summary = osfr::osf_download(project, conflict="skip")
    df = read.csv(data_summary[["local_path"]]) 
    return(df)
}
